cmake_minimum_required(VERSION 3.16)
project(furbomek_kinect2 LANGUAGES CXX)
find_package(freenect2 REQUIRED)
find_library(TURBOJPEG_LIBRARY NAMES turbojpeg REQUIRED)
add_executable(furbomek-kinect2 kinect2.cpp)
target_compile_features(furbomek-kinect2 PRIVATE cxx_std_17)
target_compile_options(furbomek-kinect2 PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_include_directories(furbomek-kinect2 PRIVATE ${freenect2_INCLUDE_DIR})
target_link_libraries(furbomek-kinect2 PRIVATE ${freenect2_LIBRARIES} ${TURBOJPEG_LIBRARY})
